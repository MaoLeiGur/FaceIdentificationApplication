# Face ID library
set(FACE_ID_DIR "${CMAKE_CURRENT_SOURCE_DIR}/code/FaceId")
set(FACE_ID_INCLUDE_DIR "${FACE_ID_DIR}/include")
set(FACE_ID_SOURCE_DIR "${FACE_ID_DIR}/source")
file(GLOB_RECURSE FACE_ID_SOURCE_FILES  "${FACE_ID_SOURCE_DIR}/*.cpp*")
file(GLOB_RECURSE FACE_ID_INCLUDE_FILES  "${FACE_ID_INCLUDE_DIR}/*.hpp*")

# Method 1: Print entire list as one line
message(STATUS "Full list: ${FACE_ID_SOURCE_FILES}")

# Get all source files from recursive search in super folder
add_library(face_id_lib
    ${FACE_ID_SOURCE_FILES}
	${FACE_ID_INCLUDE_FILES}
)


set(INCLUDE_DIRS 
	${FACE_ID_INCLUDE_DIR}
	${JSON_DIR}/include
    ${SPDLOG_DIR}/include
)

target_link_libraries(face_id_lib
    PUBLIC
        ${OpenCV_LIBS}
        spdlog::spdlog
        #SQLite::SQLite3
        #opencv_wrapper
        #dnn_engine
        #database_engine
)

target_include_directories(face_id_lib
    PUBLIC
        ${CMAKE_SOURCE_DIR}/code/include
		${FACE_ID_INCLUDE_DIR}
        ${JSON_DIR}/include
        ${SPDLOG_DIR}/include
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
)

# Main application
add_executable(face_id_app 
    "${CMAKE_CURRENT_SOURCE_DIR}/code/main.cpp")
target_link_libraries(face_id_app PRIVATE face_id_lib spdlog::spdlog ${OpenCV_LIBS} )
